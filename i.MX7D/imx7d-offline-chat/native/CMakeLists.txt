cmake_minimum_required(VERSION 3.22)
project(imx7_offline_chat LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_APP LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER LLAMA_CURL LLAMA_OPENSSL GGML_NATIVE GGML_METAL GGML_ACCELERATE GGML_BLAS GGML_OPENMP GGML_LLAMAFILE GGML_CPU_KLEIDIAI GGML_BACKEND_DL)
 set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
get_filename_component(WORKSPACE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${WORKSPACE}/demo/android-voice-weather/third_party/llama.cpp" llama)
add_executable(imx7-local-chat local_chat.cpp)
target_link_libraries(imx7-local-chat PRIVATE llama)
